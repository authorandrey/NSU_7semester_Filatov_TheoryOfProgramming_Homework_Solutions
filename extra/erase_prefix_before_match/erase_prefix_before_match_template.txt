alphabet = [#, a, b]
tapes = [ right ]

// P = p[0]p[1]...p[L-1] — непустой шаблон над алфавитом {a,b}.
// Search_i: среди уже просмотренных символов самая длинная суффиксная
// подстрока, совпадающая с префиксом P, имеет длину i, где 0 <= i < L.
// delta(i,c) — длина самого длинного собственного префикса P, который
// является суффиксом строки P[0:i] + c.

START:
    # -> #R Search_0

// Для каждого i = 0,...,L-1:
Search_i:
    // Для каждого c из {a,b}, если c не завершает шаблон:
    c -> #R Search_delta(i,c)
    // Если i = L-1 и c = P[L-1], найдено самое левое вхождение:
    c -> cL Restore_(L-1)       // если L > 1; c = P[L-1]
    c -> cH STOP                // если L = 1; c = P[0]
    // Если достигнут правый ограничитель до совпадения, вся строка уже стёрта:
    # -> #H STOP

// Генерируются только при L > 1. Для j = L-1,...,2:
Restore_j:
    # -> P[j-1]L Restore_(j-1)
    a -> P[j-1]L Restore_(j-1)
    b -> P[j-1]L Restore_(j-1)

// Последний шаг восстановления:
Restore_1:
    # -> P[0]H STOP
    a -> P[0]H STOP
    b -> P[0]H STOP

// Правила раскрытия шаблона:
// 1. Для каждого Search_i сгенерировать переходы для a и b.
// 2. Если i = L-1 и c = P[L-1], сгенерировать только переход успеха.
// 3. Иначе заменить c на #, перейти вправо в Search_delta(i,c).
// 4. В каждом Search_i добавить # -> #H STOP.
// 5. Если L = 1, состояния Restore не нужны.
// 6. При успешном совпадении ранее стёртые L-1 символов шаблона
//    восстанавливаются справа налево. Префикс перед вхождением остаётся стёртым.
// 7. Если вхождения нет, машина доходит до правого # и останавливается;
//    вся входная строка к этому моменту стёрта.
