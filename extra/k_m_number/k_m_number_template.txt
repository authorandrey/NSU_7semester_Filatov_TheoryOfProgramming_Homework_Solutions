// Перевод числа из k-ричной системы в m-ричную.
// digit(v) — символ цифры со значением v: 0,...,9,A,...,Z (A=10,...,Z=35).
// Вход записан на первой ленте наоборот (младший разряд слева).
// Первая лента не изменяется, вторая используется как копия и очищается,
// результат записывается на третью ленту наоборот.

alphabet = [#, digit(0), ..., digit(max(k,m)-1)]
tapes = [ right, right, right ]

START:
    #, #, # -> #R, #R, #R INIT

// Для каждой цифры d=0,...,k-1 создаём первый разряд копии и результат.
INIT:
    digit(d), #, # -> digit(d)H, #H, 0H COPY_FIRST

    // Пустая запись трактуется как число 0.
    #, #, # -> #H, #H, 0H STOP

// Копируем цифры с первой ленты на вторую, не меняя первую ленту.
COPY_FIRST:
    // Для каждой цифры d=0,...,k-1
    digit(d), #, 0 -> digit(d)R, digit(d)R, 0H COPY

COPY:
    // Для каждой цифры d=0,...,k-1
    digit(d), #, 0 -> digit(d)R, digit(d)R, 0H COPY
    #, #, 0 -> #H, #L, 0H CONVERT

// Обрабатываем цифры копии от старшего разряда к младшему.
// Для каждого d=0,...,k-1 и x=0,...,m-1:
// t=k*x+d; записываем t mod m, перенос floor(t/m).
CONVERT:
    #, digit(d), digit(x) -> #H, digit(d)H, digit((k*x+d) mod m)R CARRY_{floor((k*x+d)/m)}_SRC_{d}

    // Если на второй ленте достигнута левая граница, число обработано.
    // Для каждого x=0,...,m-1
    #, #, digit(x) -> #H, #H, digit(x)L CLEAN_START

// Для каждого c=0,...,k-1 и d=0,...,k-1.
// Состояние хранит перенос c и текущую цифру входа d.
// Для каждого x=0,...,m-1:
// t=k*x+c; записываем t mod m, перенос floor(t/m).
CARRY_{c}_SRC_{d}:
    #, digit(d), digit(x) -> #H, digit(d)H, digit((k*x+c) mod m)R CARRY_{floor((k*x+c)/m)}_SRC_{d}

    // Если c=0, перенос дописывать не нужно.
    #, digit(d), # -> #H, digit(d)H, #L REWIND

    // Если c>0, дописываем его цифры в системе m.
    // q=floor(c/m). Если q=0, переходим влево к REWIND;
    // если q>0, продолжаем дописывать в FLUSH_q.
    #, digit(d), # -> #H, digit(d)H, digit(c mod m){L REWIND || R FLUSH_{floor(c/m)}}

// Для каждого q=1,...,k-1 и d=0,...,k-1 дописываем остаток переноса.
// q2=floor(q/m). При q2=0 возвращаемся в REWIND, иначе продолжаем в FLUSH_q2.
FLUSH_{q}:
    #, digit(d), # -> #H, digit(d)H, digit(q mod m){L REWIND || R FLUSH_{floor(q/m)}}

// Возвращаем головку третьей ленты к младшему разряду результата и переходим
// к следующей (более младшей) цифре на второй ленте.
REWIND:
    // Для каждого d=0,...,k-1 и x=0,...,m-1
    #, digit(d), digit(x) -> #H, digit(d)H, digit(x)L REWIND
    #, digit(d), # -> #H, digit(d)L, #R CONVERT

// После обработки всех цифр перемещаемся к началу второй ленты и очищаем её.
CLEAN_START:
    #, #, # -> #H, #R, #H CLEAN_2

CLEAN_2:
    // Для каждой цифры d=0,...,k-1
    #, digit(d), # -> #H, #R, #H CLEAN_2
    #, #, # -> #H, #H, #H STOP
